cmake_minimum_required(VERSION 3.16)
project(robocup_collision_processing)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_behaviors REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
find_path(NANOFLANN_INCLUDE_DIR nanoflann.hpp REQUIRED)
add_executable(collision_cloud_node src/collision_cloud.cpp)
target_link_libraries(collision_cloud_node Eigen3::Eigen yaml-cpp Threads::Threads)
ament_target_dependencies(collision_cloud_node rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs tf2 tf2_ros)
target_compile_options(collision_cloud_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(collision_guard_node src/collision_guard.cpp)
target_include_directories(collision_guard_node PRIVATE ${NANOFLANN_INCLUDE_DIR})
target_link_libraries(collision_guard_node Eigen3::Eigen yaml-cpp Threads::Threads nlohmann_json::nlohmann_json)
ament_target_dependencies(collision_guard_node rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs tf2 tf2_ros)
target_compile_options(collision_guard_node PRIVATE -Wall -Wextra -Wpedantic)
add_library(collision_scene_layer SHARED src/scene_layer.cpp src/route_avoidance_layer.cpp)
target_link_libraries(collision_scene_layer Eigen3::Eigen nlohmann_json::nlohmann_json)
ament_target_dependencies(collision_scene_layer rclcpp sensor_msgs std_msgs tf2 tf2_ros nav2_costmap_2d pluginlib)
target_compile_options(collision_scene_layer PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(nav2_costmap_2d collision_layers.xml)
add_library(collision_guarded_behaviors SHARED src/guarded_backup.cpp)
target_link_libraries(collision_guarded_behaviors nlohmann_json::nlohmann_json)
ament_target_dependencies(collision_guarded_behaviors rclcpp std_msgs nav_msgs geometry_msgs nav2_behaviors nav2_core nav2_msgs)
target_compile_options(collision_guarded_behaviors PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(nav2_core collision_behaviors.xml)
install(TARGETS collision_cloud_node collision_guard_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS collision_scene_layer LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS collision_guarded_behaviors LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
ament_package()
